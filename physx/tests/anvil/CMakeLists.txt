cmake_minimum_required(VERSION 3.20)
project(AnvilBenchmark LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ANVIL_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../compiler/anvil/vendor" CACHE PATH "Anvil dependencies")
find_package(OpenMP REQUIRED)
set(ANVIL_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../source/lowleveldynamics/src/anvil/core")
set(ANVIL_EIGEN "${ANVIL_CORE}/../vendor/eigen")
option(PX_ANVIL_USE_AVX2 "Use AVX2 and FMA in the Anvil solver core" ON)
if(NOT TARGET PhysXAnvilCore)
	add_subdirectory("${ANVIL_CORE}" anvil-core)
endif()
add_library(AnvilSolver STATIC $<TARGET_OBJECTS:PhysXAnvilCore>)
target_include_directories(AnvilSolver PUBLIC "${ANVIL_CORE}")
add_library(AnvilTestEigen INTERFACE)
target_include_directories(AnvilTestEigen SYSTEM INTERFACE "${ANVIL_EIGEN}")
target_compile_definitions(AnvilTestEigen INTERFACE EIGEN_DONT_PARALLELIZE)
get_target_property(ANVIL_CORE_BINARY PhysXAnvilCore BINARY_DIR)
add_executable(AnvilBenchmark AnvilBenchmark.cpp)
target_link_libraries(AnvilBenchmark PRIVATE AnvilSolver AnvilTestEigen OpenMP::OpenMP_CXX)
if(MSVC)
	target_compile_options(PhysXAnvilCore PRIVATE /O2 /fp:precise /W4 /Zi)
	target_compile_options(AnvilBenchmark PRIVATE /O2 /fp:precise /W4 /Zi)
	target_link_options(AnvilBenchmark PRIVATE /INCREMENTAL:NO /DEBUG:FULL /OPT:REF /OPT:ICF)
endif()

enable_testing()
add_test(NAME AnvilMixedConstraints COMMAND AnvilBenchmark validate)

if(MSVC)
	add_library(AnvilAllocationSolver STATIC "${ANVIL_CORE}/AnvilSolver.cpp")
	add_dependencies(AnvilAllocationSolver PhysXAnvilCore)
	get_target_property(ANVIL_CORE_SOURCES PhysXAnvilCore SOURCES)
	list(FILTER ANVIL_CORE_SOURCES INCLUDE REGEX "\\.c$")
	add_library(AnvilAllocationMetis STATIC ${ANVIL_CORE_SOURCES})
	target_include_directories(AnvilAllocationMetis PRIVATE "${ANVIL_CORE}/../vendor/gklib/include"
		"${ANVIL_CORE}/../vendor/metis/include" "${ANVIL_CORE}/../vendor/metis/libmetis")
	target_compile_definitions(AnvilAllocationMetis PRIVATE NDEBUG NDEBUG2
		METIS_NodeND=Anvil_METIS_NodeND METIS_SetDefaultOptions=Anvil_METIS_SetDefaultOptions
		__thread=__declspec\(thread\))
	target_compile_options(AnvilAllocationMetis PRIVATE /O2 /W0)
	set_property(TARGET AnvilAllocationMetis PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
	target_link_libraries(AnvilAllocationSolver PRIVATE AnvilAllocationMetis)
	set_target_properties(AnvilAllocationSolver PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON)
	target_include_directories(AnvilAllocationSolver PRIVATE "${ANVIL_CORE_BINARY}"
		PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${ANVIL_CORE}")
	target_compile_definitions(AnvilAllocationSolver PUBLIC _ITERATOR_DEBUG_LEVEL=0 _CRT_SECURE_NO_WARNINGS)
	add_executable(AnvilAllocationAudit AllocationAudit.cpp)
	add_executable(PalletAllocationAudit pallet/PalletAllocationAudit.cpp)
	foreach(target AnvilAllocationSolver AnvilAllocationAudit PalletAllocationAudit)
		target_compile_options(${target} PRIVATE /O2 /fp:precise /W4 /EHsc)
		set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebugDLL)
	endforeach()
	foreach(target AnvilAllocationAudit PalletAllocationAudit)
		target_link_libraries(${target} PRIVATE AnvilAllocationSolver)
	endforeach()
	target_include_directories(PalletAllocationAudit SYSTEM PRIVATE "${ANVIL_VENDOR}/mujoco/include")
	target_link_libraries(PalletAllocationAudit PRIVATE "${ANVIL_VENDOR}/mujoco/mujoco.lib")
	add_custom_command(TARGET PalletAllocationAudit POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
		"${ANVIL_VENDOR}/mujoco/mujoco.dll" "$<TARGET_FILE_DIR:PalletAllocationAudit>")
	add_test(NAME AnvilStorageAllocations COMMAND AnvilAllocationAudit)
	add_test(NAME AnvilPalletAllocations COMMAND PalletAllocationAudit "${CMAKE_CURRENT_SOURCE_DIR}/pallet/PalletConveyor.xml")
endif()

add_executable(AnvilPileBenchmark pile/AnvilPileBenchmark.cpp)
target_include_directories(AnvilPileBenchmark PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(AnvilPileBenchmark SYSTEM PRIVATE "${ANVIL_VENDOR}/mujoco/include")
target_link_libraries(AnvilPileBenchmark PRIVATE AnvilSolver AnvilTestEigen "${ANVIL_VENDOR}/mujoco/mujoco.lib")
if(MSVC)
	target_compile_options(AnvilPileBenchmark PRIVATE /O2 /fp:precise /W4 /Zi)
	target_compile_definitions(AnvilPileBenchmark PRIVATE _CRT_SECURE_NO_WARNINGS)
	target_link_options(AnvilPileBenchmark PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
endif()
add_custom_command(TARGET AnvilPileBenchmark POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
	"${ANVIL_VENDOR}/mujoco/mujoco.dll" "$<TARGET_FILE_DIR:AnvilPileBenchmark>")

add_executable(AnvilBoundedValidation BoundedValidation.cpp)
target_link_libraries(AnvilBoundedValidation PRIVATE AnvilSolver AnvilTestEigen)
if(MSVC)
	target_compile_options(AnvilBoundedValidation PRIVATE /O2 /fp:precise /W4)
endif()
add_test(NAME AnvilBoundedRows COMMAND AnvilBoundedValidation)

add_executable(AnvilPatchValidation PatchValidation.cpp)
target_link_libraries(AnvilPatchValidation PRIVATE AnvilSolver AnvilTestEigen)
add_test(NAME AnvilPatchGroups COMMAND AnvilPatchValidation)
add_executable(AnvilPatchProjectionAudit PatchProjectionAudit.cpp)
target_include_directories(AnvilPatchProjectionAudit PRIVATE "${ANVIL_CORE}")
add_test(NAME AnvilPatchProjection COMMAND AnvilPatchProjectionAudit)

add_executable(AnvilContinuationValidation ContinuationValidation.cpp)
target_link_libraries(AnvilContinuationValidation PRIVATE AnvilSolver)
add_test(NAME AnvilContinuation COMMAND AnvilContinuationValidation)
if(MSVC)
	target_compile_options(AnvilContinuationValidation PRIVATE /O2 /fp:precise /W4)
endif()
