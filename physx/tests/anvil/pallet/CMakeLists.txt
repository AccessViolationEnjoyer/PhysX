cmake_minimum_required(VERSION 3.21)
project(PalletConveyor LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(PHYSX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(PX_GENERATE_GPU_PROJECTS OFF CACHE BOOL "" FORCE)
set(PX_GENERATE_STATIC_LIBRARIES ON CACHE BOOL "" FORCE)
set(PX_BUILDSNIPPETS OFF CACHE BOOL "" FORCE)
set(PX_BUILDPVDRUNTIME OFF CACHE BOOL "" FORCE)
add_subdirectory("${PHYSX_ROOT}" physx-sdk EXCLUDE_FROM_ALL)
add_subdirectory(.. anvil EXCLUDE_FROM_ALL)
add_executable(SnippetPalletConveyor "${PHYSX_ROOT}/snippets/snippetpalletconveyor/SnippetPalletConveyor.cpp")
target_include_directories(SnippetPalletConveyor PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../native")
target_link_libraries(SnippetPalletConveyor PRIVATE PhysXExtensions PhysX PhysXPvdSDK PhysXCommon PhysXFoundation PhysXCooking)
target_compile_definitions(SnippetPalletConveyor PRIVATE PALLET_CONVEYOR_BENCHMARK)
add_executable(MujocoPalletConveyor MujocoPalletConveyor.cpp)
target_include_directories(MujocoPalletConveyor PRIVATE .. "${ANVIL_VENDOR}/mujoco/include")
target_link_libraries(MujocoPalletConveyor PRIVATE AnvilSolver "${ANVIL_VENDOR}/mujoco/mujoco.lib")
if(MSVC)
	target_compile_options(SnippetPalletConveyor PRIVATE /O2 /W4 /fp:precise /Zi)
	target_link_options(SnippetPalletConveyor PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
	target_compile_options(MujocoPalletConveyor PRIVATE /O2 /W4 /fp:precise /Zi)
	target_link_options(MujocoPalletConveyor PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
	target_compile_definitions(SnippetPalletConveyor PRIVATE _CRT_SECURE_NO_WARNINGS PX_PHYSX_STATIC_LIB)
	target_compile_definitions(MujocoPalletConveyor PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_custom_command(TARGET MujocoPalletConveyor POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
	"${ANVIL_VENDOR}/mujoco/mujoco.dll" "$<TARGET_FILE_DIR:MujocoPalletConveyor>")

# Individual cases on ten long conveyors (../cases): PhysX PGS or Anvil, and MuJoCo.
add_executable(CaseConveyor ../cases/CaseConveyor.cpp)
target_include_directories(CaseConveyor PRIVATE ../cases ../native)
target_link_libraries(CaseConveyor PRIVATE PhysXExtensions PhysX PhysXPvdSDK PhysXCommon PhysXFoundation PhysXCooking)
# The pinned MuJoCo runs the case scene too; the latest release, when present in the vendor
# directory, builds to its own folder because both versions ship mujoco.dll.
add_executable(MujocoCaseConveyor313 ../cases/MujocoCaseConveyor.cpp)
target_include_directories(MujocoCaseConveyor313 PRIVATE ../cases .. "${ANVIL_VENDOR}/mujoco/include")
target_link_libraries(MujocoCaseConveyor313 PRIVATE "${ANVIL_VENDOR}/mujoco/mujoco.lib")
# Totes of four heavy boxes on the same conveyors (../cases): PhysX PGS or Anvil, and MuJoCo.
add_executable(ToteConveyor ../cases/ToteConveyor.cpp)
target_include_directories(ToteConveyor PRIVATE ../cases ../native)
target_link_libraries(ToteConveyor PRIVATE PhysXExtensions PhysX PhysXPvdSDK PhysXCommon PhysXFoundation PhysXCooking)
set(ANVIL_MUJOCO_LATEST "${ANVIL_VENDOR}/mujoco-3.14.0")
set(CASE_TARGETS CaseConveyor MujocoCaseConveyor313 ToteConveyor)
if(EXISTS "${ANVIL_MUJOCO_LATEST}/lib/mujoco.lib")
	foreach(scene Case Tote)
		add_executable(Mujoco${scene}Conveyor ../cases/Mujoco${scene}Conveyor.cpp)
		target_include_directories(Mujoco${scene}Conveyor PRIVATE ../cases .. "${ANVIL_MUJOCO_LATEST}/include")
		target_link_libraries(Mujoco${scene}Conveyor PRIVATE "${ANVIL_MUJOCO_LATEST}/lib/mujoco.lib")
		set_target_properties(Mujoco${scene}Conveyor PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mujoco-3.14.0")
		add_custom_command(TARGET Mujoco${scene}Conveyor POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
			"${ANVIL_MUJOCO_LATEST}/bin/mujoco.dll" "$<TARGET_FILE_DIR:Mujoco${scene}Conveyor>")
		list(APPEND CASE_TARGETS Mujoco${scene}Conveyor)
	endforeach()
	# The pallet and pile comparisons against the latest release (their pinned builds use 3.13).
	add_executable(MujocoPalletConveyorLatest MujocoPalletConveyor.cpp)
	target_include_directories(MujocoPalletConveyorLatest PRIVATE .. "${ANVIL_MUJOCO_LATEST}/include")
	target_link_libraries(MujocoPalletConveyorLatest PRIVATE AnvilSolver "${ANVIL_MUJOCO_LATEST}/lib/mujoco.lib")
	add_executable(MujocoPileBenchmarkLatest ../pile/AnvilPileBenchmark.cpp)
	target_include_directories(MujocoPileBenchmarkLatest PRIVATE .. "${ANVIL_MUJOCO_LATEST}/include")
	target_link_libraries(MujocoPileBenchmarkLatest PRIVATE AnvilSolver AnvilTestEigen "${ANVIL_MUJOCO_LATEST}/lib/mujoco.lib")
	foreach(target MujocoPalletConveyorLatest MujocoPileBenchmarkLatest)
		set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mujoco-3.14.0")
		list(APPEND CASE_TARGETS ${target})
	endforeach()
	add_executable(MujocoPalletPlatform MujocoPalletPlatform.cpp)
	target_include_directories(MujocoPalletPlatform PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" .. "${ANVIL_MUJOCO_LATEST}/include")
	target_link_libraries(MujocoPalletPlatform PRIVATE "${ANVIL_MUJOCO_LATEST}/lib/mujoco.lib")
	set_target_properties(MujocoPalletPlatform PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mujoco-3.14.0")
	add_custom_command(TARGET MujocoPalletPlatform POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
		"${ANVIL_MUJOCO_LATEST}/bin/mujoco.dll" "$<TARGET_FILE_DIR:MujocoPalletPlatform>")
	list(APPEND CASE_TARGETS MujocoPalletPlatform)
endif()
if(MSVC)
	foreach(target ${CASE_TARGETS})
		target_compile_options(${target} PRIVATE /O2 /W4 /fp:precise /EHsc /Zi)
		target_link_options(${target} PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
		target_compile_definitions(${target} PRIVATE PX_PHYSX_STATIC_LIB _CRT_SECURE_NO_WARNINGS)
	endforeach()
endif()
