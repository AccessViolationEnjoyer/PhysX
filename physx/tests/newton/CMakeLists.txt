cmake_minimum_required(VERSION 3.20)
project(NewtonBenchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(NEWTON_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../compiler/newton/vendor" CACHE PATH "Newton dependencies")
find_package(OpenMP REQUIRED)
set(NEWTON_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../source/lowleveldynamics/src/newton/core")
set(NEWTON_EIGEN "${NEWTON_CORE}/../vendor/eigen")
if(NOT TARGET PhysXNewtonCore)
	add_subdirectory("${NEWTON_CORE}" newton-core)
endif()
add_library(NewtonSolver STATIC $<TARGET_OBJECTS:PhysXNewtonCore>)
target_include_directories(NewtonSolver PUBLIC "${NEWTON_CORE}")
target_include_directories(NewtonSolver SYSTEM PUBLIC "${NEWTON_EIGEN}")
target_compile_definitions(NewtonSolver PUBLIC EIGEN_DONT_PARALLELIZE)
get_target_property(NEWTON_CORE_BINARY PhysXNewtonCore BINARY_DIR)
add_executable(NewtonBenchmark NewtonBenchmark.cpp)
target_link_libraries(NewtonBenchmark PRIVATE NewtonSolver OpenMP::OpenMP_CXX)
if(MSVC)
	target_compile_options(PhysXNewtonCore PRIVATE /O2 /fp:precise /W4 /Zi)
	target_compile_options(NewtonBenchmark PRIVATE /O2 /fp:precise /W4 /Zi)
	target_link_options(NewtonBenchmark PRIVATE /INCREMENTAL:NO /DEBUG:FULL /OPT:REF /OPT:ICF)
endif()

enable_testing()
add_test(NAME NewtonMixedConstraints COMMAND NewtonBenchmark validate)

if(MSVC)
	add_library(NewtonAllocationSolver STATIC "${NEWTON_CORE}/NewtonSolver.cpp")
	add_dependencies(NewtonAllocationSolver PhysXNewtonCore)
	set_target_properties(NewtonAllocationSolver PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON)
	target_include_directories(NewtonAllocationSolver PRIVATE "${NEWTON_CORE_BINARY}" PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${NEWTON_CORE}")
	target_include_directories(NewtonAllocationSolver SYSTEM PUBLIC "${NEWTON_EIGEN}")
	target_compile_definitions(NewtonAllocationSolver PUBLIC EIGEN_DONT_PARALLELIZE _ITERATOR_DEBUG_LEVEL=0 _CRT_SECURE_NO_WARNINGS)
	add_executable(NewtonAllocationAudit AllocationAudit.cpp)
	add_executable(PalletAllocationAudit pallet/PalletAllocationAudit.cpp)
	foreach(target NewtonAllocationSolver NewtonAllocationAudit PalletAllocationAudit)
		target_compile_options(${target} PRIVATE /O2 /fp:precise /W4 /EHsc)
		set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebugDLL)
	endforeach()
	foreach(target NewtonAllocationAudit PalletAllocationAudit)
		target_link_libraries(${target} PRIVATE NewtonAllocationSolver)
	endforeach()
	target_include_directories(PalletAllocationAudit SYSTEM PRIVATE "${NEWTON_VENDOR}/mujoco/include")
	target_link_libraries(PalletAllocationAudit PRIVATE "${NEWTON_VENDOR}/mujoco/mujoco.lib")
	add_custom_command(TARGET PalletAllocationAudit POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
		"${NEWTON_VENDOR}/mujoco/mujoco.dll" "$<TARGET_FILE_DIR:PalletAllocationAudit>")
	add_test(NAME NewtonStorageAllocations COMMAND NewtonAllocationAudit)
	add_test(NAME NewtonPalletAllocations COMMAND PalletAllocationAudit "${CMAKE_CURRENT_SOURCE_DIR}/pallet/PalletConveyor.xml")
endif()

add_executable(NewtonPileBenchmark pile/NewtonPileBenchmark.cpp)
target_include_directories(NewtonPileBenchmark PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(NewtonPileBenchmark SYSTEM PRIVATE "${NEWTON_VENDOR}/mujoco/include")
target_link_libraries(NewtonPileBenchmark PRIVATE NewtonSolver "${NEWTON_VENDOR}/mujoco/mujoco.lib")
if(MSVC)
	target_compile_options(NewtonPileBenchmark PRIVATE /O2 /fp:precise /W4 /Zi)
	target_compile_definitions(NewtonPileBenchmark PRIVATE _CRT_SECURE_NO_WARNINGS)
	target_link_options(NewtonPileBenchmark PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
endif()
add_custom_command(TARGET NewtonPileBenchmark POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
	"${NEWTON_VENDOR}/mujoco/mujoco.dll" "$<TARGET_FILE_DIR:NewtonPileBenchmark>")

add_executable(NewtonBoundedValidation BoundedValidation.cpp)
target_link_libraries(NewtonBoundedValidation PRIVATE NewtonSolver)

	target_compile_options(NewtonBoundedValidation PRIVATE /O2 /fp:precise /W4)
