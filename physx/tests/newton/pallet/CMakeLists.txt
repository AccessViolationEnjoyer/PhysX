cmake_minimum_required(VERSION 3.21)
project(PalletConveyor LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(PHYSX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(PX_GENERATE_GPU_PROJECTS OFF CACHE BOOL "" FORCE)
set(PX_GENERATE_STATIC_LIBRARIES ON CACHE BOOL "" FORCE)
set(PX_BUILDSNIPPETS OFF CACHE BOOL "" FORCE)
set(PX_BUILDPVDRUNTIME OFF CACHE BOOL "" FORCE)
add_subdirectory("${PHYSX_ROOT}" physx-sdk EXCLUDE_FROM_ALL)
add_subdirectory(.. newton EXCLUDE_FROM_ALL)
add_executable(SnippetPalletConveyor "${PHYSX_ROOT}/snippets/snippetpalletconveyor/SnippetPalletConveyor.cpp")
target_include_directories(SnippetPalletConveyor PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../native")
target_link_libraries(SnippetPalletConveyor PRIVATE PhysXExtensions PhysX PhysXPvdSDK PhysXCommon PhysXFoundation PhysXCooking)
target_compile_definitions(SnippetPalletConveyor PRIVATE PALLET_CONVEYOR_BENCHMARK)
add_executable(MujocoPalletConveyor MujocoPalletConveyor.cpp)
target_include_directories(MujocoPalletConveyor PRIVATE .. "${NEWTON_VENDOR}/mujoco/include")
target_link_libraries(MujocoPalletConveyor PRIVATE NewtonSolver "${NEWTON_VENDOR}/mujoco/mujoco.lib")
if(MSVC)
	target_compile_options(SnippetPalletConveyor PRIVATE /O2 /W4 /fp:precise /Zi)
	target_link_options(SnippetPalletConveyor PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
	target_compile_options(MujocoPalletConveyor PRIVATE /O2 /W4 /fp:precise /Zi)
	target_link_options(MujocoPalletConveyor PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
	target_compile_definitions(SnippetPalletConveyor PRIVATE _CRT_SECURE_NO_WARNINGS PX_PHYSX_STATIC_LIB)
	target_compile_definitions(MujocoPalletConveyor PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_custom_command(TARGET MujocoPalletConveyor POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
	"${NEWTON_VENDOR}/mujoco/mujoco.dll" "$<TARGET_FILE_DIR:MujocoPalletConveyor>")

# Individual cases on ten long conveyors (../cases): PhysX PGS or Newton, and MuJoCo.
add_executable(CaseConveyor ../cases/CaseConveyor.cpp)
target_include_directories(CaseConveyor PRIVATE ../cases ../native)
target_link_libraries(CaseConveyor PRIVATE PhysXExtensions PhysX PhysXPvdSDK PhysXCommon PhysXFoundation PhysXCooking)
# The pinned MuJoCo runs the case scene too; the latest release, when present in the vendor
# directory, builds to its own folder because both versions ship mujoco.dll.
add_executable(MujocoCaseConveyor313 ../cases/MujocoCaseConveyor.cpp)
target_include_directories(MujocoCaseConveyor313 PRIVATE ../cases .. "${NEWTON_VENDOR}/mujoco/include")
target_link_libraries(MujocoCaseConveyor313 PRIVATE "${NEWTON_VENDOR}/mujoco/mujoco.lib")
set(NEWTON_MUJOCO_LATEST "${NEWTON_VENDOR}/mujoco-3.14.0")
set(CASE_TARGETS CaseConveyor MujocoCaseConveyor313)
if(EXISTS "${NEWTON_MUJOCO_LATEST}/lib/mujoco.lib")
	add_executable(MujocoCaseConveyor ../cases/MujocoCaseConveyor.cpp)
	target_include_directories(MujocoCaseConveyor PRIVATE ../cases .. "${NEWTON_MUJOCO_LATEST}/include")
	target_link_libraries(MujocoCaseConveyor PRIVATE "${NEWTON_MUJOCO_LATEST}/lib/mujoco.lib")
	set_target_properties(MujocoCaseConveyor PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mujoco-3.14.0")
	add_custom_command(TARGET MujocoCaseConveyor POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
		"${NEWTON_MUJOCO_LATEST}/bin/mujoco.dll" "$<TARGET_FILE_DIR:MujocoCaseConveyor>")
	list(APPEND CASE_TARGETS MujocoCaseConveyor)
endif()
if(MSVC)
	foreach(target ${CASE_TARGETS})
		target_compile_options(${target} PRIVATE /O2 /W4 /fp:precise /EHsc /Zi)
		target_link_options(${target} PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
		target_compile_definitions(${target} PRIVATE PX_PHYSX_STATIC_LIB _CRT_SECURE_NO_WARNINGS)
	endforeach()
endif()
